/* ============================================================
   PRACTICAL 4 : Hill Cipher
   ============================================================ */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
int main()
{
    int choice;
    char cleanText[2000], resultText[2000];
    int matrix[10][10], invMatrix[10][10];
    int n, i, j, k, cleanLen = 0, ch;
    printf("=== HILL CIPHER PROGRAM (NO FUNCTIONS) ===\n");
    printf("1. Encrypt\n");
    printf("2. Decrypt\n");
    printf("Select an option (1 or 2): ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    {
        printf("Invalid choice! Program exiting.\n");
        return 1;
    }
    FILE *fp = fopen("hillinput.txt", "r");
    if (fp == NULL)
    {
        printf("\n[ERROR] File 'hillinput.txt' not found!\n");
        return 1;
    }
    while ((ch = fgetc(fp)) != EOF && cleanLen < 1990)
    {
        if (isalpha(ch))
        {
            cleanText[cleanLen++] = toupper((unsigned char)ch);
        }
    }
    fclose(fp);
    cleanText[cleanLen] = '\0';
    if (cleanLen == 0)
    {
        printf("\n[ERROR] 'hillinput.txt' contains no valid alphabetic letters!\n");
        return 1;
    }
    printf("\nCleaned Input Text: %s\n", cleanText);
    printf("Enter matrix size N (2 or 3): ");
    if (scanf("%d", &n) != 1 || (n != 2 && n != 3))
    {
        printf("[ERROR] This single-main version supports N=2 or N=3.\n");
        return 1;
    }
    printf("\nEnter %dx%d Key Matrix elements row by row:\n", n, n);
    for (i = 0; i < n; i++)
    {
        printf("Row %d values (%d numbers): ", i + 1, n);
        for (j = 0; j < n; j++)
        {
            scanf("%d", &matrix[i][j]);
            matrix[i][j] = (matrix[i][j] % 26 + 26) % 26;
        }
    }
    while (cleanLen % n != 0)
    {
        cleanText[cleanLen++] = 'X';
    }
    cleanText[cleanLen] = '\0';
    printf("Padded Text for Processing: %s\n", cleanText);
    if (choice == 1)
    {
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += matrix[j][k] * val;
                }
                resultText[resIdx++] = ((sum % 26) + 26) % 26 + 'A';
            }
        }
        resultText[resIdx] = '\0';
        FILE *outFp = fopen("output.txt", "w");
        if (outFp != NULL)
        {
            fputs(resultText, outFp);
            fclose(outFp);
        }
        printf("\n-------------------------\n");
        printf("Encryption Successful!\n");
        printf("Generated Ciphertext: %s\n", resultText);
        printf("Result saved to 'output.txt'\n");
        printf("-------------------------\n");
    }
    else if (choice == 2)
    {
        int det = 0;
        if (n == 2)
        {
            det = matrix[0][0] * matrix[1][1] - matrix[0][1] * matrix[1][0];
        }
        else if (n == 3)
        {
            det = matrix[0][0] * (matrix[1][1] * matrix[2][2] - matrix[1][2] * matrix[2][1]) -
                  matrix[0][1] * (matrix[1][0] * matrix[2][2] - matrix[1][2] * matrix[2][0]) +
                  matrix[0][2] * (matrix[1][0] * matrix[2][1] - matrix[1][1] * matrix[2][0]);
        }
        det = (det % 26 + 26) % 26;
        int detInv = -1;
        for (int x = 1; x < 26; x++)
        {
            if ((det * x) % 26 == 1)
            {
                detInv = x;
                break;
            }
        }
        if (detInv == -1)
        {
            printf("\n[ERROR] Key matrix is not invertible modulo 26!\n");
            return 1;
        }
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
            {
                int subDet = 0;
                if (n == 2)
                {
                    int r = (i == 0) ? 1 : 0;
                    int c = (j == 0) ? 1 : 0;
                    subDet = matrix[r][c];
                }
                else if (n == 3)
                {
                    int sub[2][2];
                    int r_idx = 0;
                    for (int row = 0; row < 3; row++)
                    {
                        if (row == i) continue;
                        int c_idx = 0;
                        for (int col = 0; col < 3; col++)
                        {
                            if (col == j) continue;
                            sub[r_idx][c_idx++] = matrix[row][col];
                        }
                        r_idx++;
                    }
                    subDet = sub[0][0] * sub[1][1] - sub[0][1] * sub[1][0];
                }
                int sign = ((i + j) % 2 == 0) ? 1 : -1;
                int cofactor = sign * subDet;
                cofactor = (cofactor % 26 + 26) % 26;
                invMatrix[j][i] = (cofactor * detInv) % 26;
            }
        }
        printf("\nCalculated Inverse Key Matrix (mod 26):\n");
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
            {
                printf("%2d ", invMatrix[i][j]);
            }
            printf("\n");
        }
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += invMatrix[j][k] * val;
                }
                resultText[resIdx++] = ((sum % 26) + 26) % 26 + 'A';
            }
        }
        resultText[resIdx] = '\0';
        FILE *outFp = fopen("output.txt", "w");
        if (outFp != NULL)
        {
            fputs(resultText, outFp);
            fclose(outFp);
        }
        printf("\n-------------------------\n");
        printf("Decryption Successful!\n");
        printf("Generated Plaintext: %s\n", resultText);
        printf("Result saved to 'output.txt'\n");
        printf("-------------------------\n");
    }
    return 0;
}


/* ============================================================
   PRACTICAL 5 : Vigenere Cipher (with Autokey option)
   ============================================================ */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#define MAX_LEN 1000
int main()
{
    char input_text[MAX_LEN] = {0};
    char initial_key[MAX_LEN] = {0};
    char adjusted_key[MAX_LEN] = {0};
    char output_text[MAX_LEN] = {0};
    int mode_choice, key_choice;
    int text_len, k_len, i;
    FILE *fin, *fout;
    printf("--- Cryptography System ---\n1. Encrypt\n2. Decrypt\nChoice: ");
    if (scanf("%d", &mode_choice) != 1) return 1;
    fin = fopen("input.txt", "r");
    if (fin == NULL)
    {
        printf("Error: Could not open 'input.txt'. Please create it first!\n");
        return 1;
    }
    fgets(input_text, MAX_LEN, fin);
    input_text[strcspn(input_text, "\r\n")] = 0;
    fclose(fin);
    text_len = strlen(input_text);
    if (text_len == 0)
    {
        printf("Error: 'input.txt' is empty.\n");
        return 1;
    }
    printf("\nText read from file: %s\n", input_text);
    printf("Enter the secret key: ");
    scanf("%s", initial_key);
    k_len = strlen(initial_key);
    printf("\n--- Keystream Adjustment Menu ---\n");
    printf("1. Repeat the key to match length (Vigenere style)\n");
    printf("2. Use Autokey style to match length\n");
    printf("Enter choice (1 or 2): ");
    scanf("%d", &key_choice);
    if (key_choice == 1)
    {
        for (i = 0; i < text_len; i++)
        {
            adjusted_key[i] = initial_key[i % k_len];
        }
        adjusted_key[text_len] = '\0';
    }
    else if (key_choice == 2)
    {
        for (i = 0; i < k_len; i++)
        {
            adjusted_key[i] = initial_key[i];
        }
        if (mode_choice == 1)
        {
            for (i = k_len; i < text_len; i++)
            {
                adjusted_key[i] = input_text[i - k_len];
            }
        }
        else
        {
            for (i = k_len; i < text_len; i++)
            {
                adjusted_key[i] = '\0';
            }
        }
        adjusted_key[text_len] = '\0';
    }
    else
    {
        printf("Invalid choice! Exiting program.\n");
        return 1;
    }
    for (i = 0; i < text_len; i++)
    {
        if (isalpha(input_text[i]))
        {
            char t_base = isupper(input_text[i]) ? 'A' : 'a';
            char k_base = isupper(adjusted_key[i]) ? 'A' : 'a';
            int t_val = input_text[i] - t_base;
            int k_val = adjusted_key[i] - k_base;
            if (mode_choice == 1)
            {
                output_text[i] = ((t_val + k_val) % 26) + t_base;
            }
            else
            {
                output_text[i] = ((t_val - k_val + 26) % 26) + t_base;
                if (key_choice == 2 && (i + k_len < text_len))
                {
                    adjusted_key[i + k_len] = output_text[i];
                }
            }
        }
        else
        {
            output_text[i] = input_text[i];
            if (mode_choice == 2 && key_choice == 2 && (i + k_len < text_len))
            {
                adjusted_key[i + k_len] = input_text[i];
            }
        }
    }
    output_text[text_len] = '\0';
    fout = fopen("output.txt", "w");
    if (fout == NULL)
    {
        printf("Error: Could not create 'output.txt'.\n");
        return 1;
    }
    if (mode_choice == 1)
    {
        fprintf(fout, "Ciphertext: %s\n", output_text);
    }
    else
    {
        fprintf(fout, "Plaintext: %s\n", output_text);
    }
    fprintf(fout, "Used Keystream: %s\n", adjusted_key);
    fclose(fout);
    printf("\nOperation successful! Results saved to 'output.txt'.\n");
    return 0;
}
