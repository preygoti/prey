#include <ctype.h>

'S', 'T', 'U', 'V', 'W', 'X', 'Y', 'Z'};
const char KEY[] = {'M', 'U', 'A', 'L', 'V', 'O', 'Z', 'K', 'R', 'N', 'J', 'X', 'Q', 'D', 'F', 'S', 'H', 'P', 'E', 'B',

C Practical 3: Playfair Cipher — 5×5 Matrix Cipher

#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main()
{
    char key[100], text[500];
    char matrix[5][5];
    int used[26] = {0};
    int i, j, k = 0;
    printf("Enter Key: ");
    scanf("%s", key);
    used['J' - 'A'] = 1;
    // Generate Matrix
    for (i = 0; key[i] != '\0'; i++)
    {
        char ch = toupper(key[i]);
        if (ch == 'J') ch = 'T';
        if (ch >= 'A' && ch <= 'Z') { }
        if (used[ch - 'A'] == 0) { }
        matrix[k / 5][k % 5] = ch;
        used[ch - 'A'] = 1;
        k++;
    }
    for (i = 0; i < 26; i++)
    {
        if (used[i] == 0) { matrix[k/5][k%5] = i + 'A'; k++; }
    }
    printf("\nKey Matrix\n\n");
    for (i = 0; i < 5; i++)
    {
        for (j = 0; j < 5; j++) printf("%c", matrix[i][j]);
        printf("\n");
    }
    FILE *fp, *fe, *fd;
    fp = fopen("file1.txt", "r");
    if (fp == NULL) { printf("File not found"); return 0; }
    fgets(text, 500, fp); fclose(fp);
    char plain[500]; int p = 0;
    for (i = 0; text[i] != '\0'; i++)
    {
        if (isalpha(text[i]))
        { plain[p] = toupper(text[i]); if (plain[p]=='J') plain[p]='T'; p++; }
    }
    plain[p] = '\0';
    char final[500]; int f = 0;
    for (i = 0; i < p; i++)
    {
        final[f++] = plain[i];
        if (i < p-1 && plain[i] == plain[i+1]) final[f++] = 'X';
    }
    if (f % 2 != 0) final[f++] = 'X';
    final[f] = '\0';
    printf("\nPlain Text: %s\n", final);
    // Encryption
    fe = fopen("file2.txt", "w");
    char cipher[500]; int c = 0;
    for (i = 0; i < f; i += 2)
    {
        char a = final[i], b = final[i+1];
        int r1, c1, r2, c2;
        for (int x=0; x<5; x++)
          for (int y=0; y<5; y++)
          { if(matrix[x][y]==a){r1=x;c1=y;} if(matrix[x][y]==b){r2=x;c2=y;} }
        if (r1==r2) { cipher[c++]=matrix[r1][(c1+1)%5]; cipher[c++]=matrix[r2][(c2+1)%5]; }
        else if (c1==c2) { cipher[c++]=matrix[(r1+1)%5][c1]; cipher[c++]=matrix[(r2+1)%5][c2]; }
        else { cipher[c++]=matrix[r1][c2]; cipher[c++]=matrix[r2][c1]; }
    }
    cipher[c] = '\0';
    fprintf(fe, "%s", cipher); fclose(fe);
    printf("Cipher Text: %s\n", cipher);
    // Decryption
    fe = fopen("file2.txt", "r");
    if (fe == NULL) { printf("Encrypted file not found"); return 0; }
    fgets(cipher, 500, fe); fclose(fe);
    c = strlen(cipher);
    fd = fopen("decrypted.txt", "w");
    char decrypt[500]; int d = 0;
    for (i = 0; i < c; i += 2)
    {
        char a = cipher[i], b = cipher[i+1];
        int r1, c1, r2, c2;
        for (int x=0; x<5; x++)
          for (int y=0; y<5; y++)
          { if(matrix[x][y]==a){r1=x;c1=y;} if(matrix[x][y]==b){r2=x;c2=y;} }
        if (r1==r2) { decrypt[d++]=matrix[r1][(c1+4)%5]; decrypt[d++]=matrix[r2][(c2+4)%5]; }
        else if (c1==c2) { decrypt[d++]=matrix[(r1+4)%5][c1]; decrypt[d++]=matrix[(r2+4)%5][c2]; }
        else { decrypt[d++]=matrix[r1][c2]; decrypt[d++]=matrix[r2][c1]; }
    }
    decrypt[d] = '\0';
    fprintf(fd, "%s", decrypt); fclose(fd);
    printf("Decrypted Text: %s\n", decrypt);
    return 0;
}

Hill Cipher

C Practical 4: Hill Cipher — Matrix-Based Encryption

/* Full Hill Cipher - Encrypt & Decrypt with Matrix Inverse */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
int main()
{
    int choice;
    char cleanText[2000], resultText[2000];
    int matrix[10][10], invMatrix[10][10];
    int n, i, j, k, cleanLen = 0, ch;
    printf("=== HILL CIPHER PROGRAM ===\n1. Encrypt\n2. Decrypt\nSelect: ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    { printf("Invalid!\n"); return 1; }
    FILE *fp = fopen("hillinput.txt", "r");
    if (!fp) { printf("File not found!\n"); return 1; }
    while ((ch=fgetc(fp))!=EOF && cleanLen<1990)
        if (isalpha(ch)) cleanText[cleanLen++] = toupper(ch);
    fclose(fp); cleanText[cleanLen]='\0';
    if (!cleanLen) { printf("No letters!\n"); return 1; }
    printf("Text: %s\nMatrix size N (2/3): ", cleanText);
    if (scanf("%d",&n)!=1||(n!=2&&n!=3)) { printf("N=2 or 3 only\n"); return 1; }
    printf("Enter %dx%d matrix:\n", n, n);
    for(i=0;i<n;i++) for(j=0;j<n;j++)
    { scanf("%d",&matrix[i][j]); matrix[i][j]=(matrix[i][j]%26+26)%26; }
    while(cleanLen%n) cleanText[cleanLen++]='X';
    cleanText[cleanLen]='\0';
    if (choice==1) {
        int r=0;
        for(i=0;i<cleanLen;i+=n) for(j=0;j<n;j++)
        { int s=0; for(k=0;k<n;k++) s+=matrix[j][k]*(cleanText[i+k]-'A');
          resultText[r++]=((s%26)+26)%26+'A'; }
        resultText[r]='\0';
        FILE *o=fopen("output.txt","w"); if(o){fputs(resultText,o);fclose(o);}
        printf("Encrypted: %s\n", resultText);
    } else {
        int det=0;
        if(n==2) det=matrix[0][0]*matrix[1][1]-matrix[0][1]*matrix[1][0];
        else det=matrix[0][0]*(matrix[1][1]*matrix[2][2]-matrix[1][2]*matrix[2][1])
              -matrix[0][1]*(matrix[1][0]*matrix[2][2]-matrix[1][2]*matrix[2][0])
              +matrix[0][2]*(matrix[1][0]*matrix[2][1]-matrix[1][1]*matrix[2][0]);
        det=(det%26+26)%26;
        int detInv=-1;
        for(int x=1;x<26;x++) if((det*x)%26==1){detInv=x;break;}
        if(detInv==-1){printf("Not invertible!\n");return 1;}
        for(i=0;i<n;i++) for(j=0;j<n;j++) {
            int sd=0;
            if(n==2){int r=(i==0)?1:0,c=(j==0)?1:0; sd=matrix[r][c];}
            else{int sub[2][2],ri=0;
              for(int row=0;row<3;row++){if(row==i)continue;int ci=0;
                for(int col=0;col<3;col++){if(col==j)continue;sub[ri][ci++]=matrix[row][col];}ri++;}
              sd=sub[0][0]*sub[1][1]-sub[0][1]*sub[1][0];}
            int cof=((i+j)%2==0?1:-1)*sd;
            invMatrix[j][i]=((cof%26+26)%26*detInv)%26;
        }
        int r=0;
        for(i=0;i<cleanLen;i+=n) for(j=0;j<n;j++)
        { int s=0; for(k=0;k<n;k++) s+=invMatrix[j][k]*(cleanText[i+k]-'A');
          resultText[r++]=((s%26)+26)%26+'A'; }
        resultText[r]='\0';
        FILE *o=fopen("output.txt","w"); if(o){fputs(resultText,o);fclose(o);}
        printf("Decrypted: %s\n", resultText);
    }
    return 0;
}

Vigenere Cipher

C Practical 5: Vigenere Cipher (with Autokey Option)

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#define MAX_LEN 1000
int main()
{
    char input_text[MAX_LEN]={0}, initial_key[MAX_LEN]={0};
    char adjusted_key[MAX_LEN]={0}, output_text[MAX_LEN]={0};
    int mode_choice, key_choice, text_len, k_len, i;
    FILE *fin, *fout;
    printf("--- Cryptography System ---\n1. Encrypt\n2. Decrypt\nChoice: ");
    if (scanf("%d", &mode_choice) != 1) return 1;
    fin = fopen("input.txt", "r");
    if (!fin) { printf("Cannot open input.txt!\n"); return 1; }
    fgets(input_text, MAX_LEN, fin);
    input_text[strcspn(input_text, "\r\n")] = 0;
    fclose(fin);
    text_len = strlen(input_text);
    if (!text_len) { printf("Empty file!\n"); return 1; }
    printf("Text: %s\nKey: ", input_text);
    scanf("%s", initial_key);
    k_len = strlen(initial_key);
    printf("1. Vigenere repeat\n2. Autokey\nChoice: ");
    scanf("%d", &key_choice);
    if (key_choice==1)
    { for(i=0;i<text_len;i++) adjusted_key[i]=initial_key[i%k_len]; }
    else if (key_choice==2)
    {
        for(i=0;i<k_len;i++) adjusted_key[i]=initial_key[i];
        if(mode_choice==1)
            for(i=k_len;i<text_len;i++) adjusted_key[i]=input_text[i-k_len];
        else
            for(i=k_len;i<text_len;i++) adjusted_key[i]='\0';
    }
    else { printf("Invalid!\n"); return 1; }
    adjusted_key[text_len]='\0';
    for (i=0; i<text_len; i++)
    {
        if (isalpha(input_text[i]))
        {
            char tb=isupper(input_text[i])?'A':'a';
            char kb=isupper(adjusted_key[i])?'A':'a';
            int tv=input_text[i]-tb, kv=adjusted_key[i]-kb;
            if(mode_choice==1)
                output_text[i]=((tv+kv)%26)+tb;
            else {
                output_text[i]=((tv-kv+26)%26)+tb;
                if(key_choice==2 && i+k_len<text_len)
                    adjusted_key[i+k_len]=output_text[i];
            }
        } else {
            output_text[i]=input_text[i];
            if(mode_choice==2 && key_choice==2 && i+k_len<text_len)
                adjusted_key[i+k_len]=input_text[i];
        }
    }
    output_text[text_len]='\0';
    fout=fopen("output.txt","w");
    if(mode_choice==1) fprintf(fout,"Ciphertext: %s\n",output_text);
    else fprintf(fout,"Plaintext: %s\n",output_text);
    fprintf(fout,"Keystream: %s\n",adjusted_key);
    fclose(fout);
    printf("Done! Check output.txt\n");
    return 0;
}

C Practical 6: Columnar Transposition Cipher

#include <stdio.h>
#include <string.h>
int main()
{
    int choice, i, j, k, rows, n;
    char text[100], key[20], result[100];
    char matrix[100][20];
    FILE *input, *output;
    printf("--- Columnar Transposition ---\n1. Encrypt\n2. Decrypt\nChoice: ");
    scanf("%d", &choice);
    printf("Key: "); scanf("%s", key);
    n = strlen(key);
    input = fopen("input.txt", "r");
    if (!input) { printf("File not found!"); return 0; }
    output = fopen("output.txt", "w");
    while (fgets(text, sizeof(text), input))
    {
        text[strcspn(text, "\n")] = '\0';
        if (!strlen(text)) continue;
        if (choice == 1)
        {
            while (strlen(text) % n) strcat(text, "X");
            rows = strlen(text) / n;
            k = 0;
            for(i=0;i<rows;i++) for(j=0;j<n;j++) matrix[i][j]=text[k++];
            k = 0;
            for(j=0;j<n;j++) for(i=0;i<rows;i++) result[k++]=matrix[i][j];
            result[k] = '\0';
        }
        else if (choice == 2)
        {
            rows = strlen(text) / n;
            k = 0;
            for(j=0;j<n;j++) for(i=0;i<rows;i++) matrix[i][j]=text[k++];
            k = 0;
            for(i=0;i<rows;i++) for(j=0;j<n;j++) result[k++]=matrix[i][j];
            result[k] = '\0';
        }
        else { printf("Invalid!"); return 0; }
        fprintf(output, "%s\n", result);
    }
    fclose(input); fclose(output);
    printf("Done! Check output.txt\n");
    return 0;
}

Rail Fence Transposition Cipher

C Practical 7: Rail Fence Transposition Cipher

#include <stdio.h>
#include <string.h>
int main()
{
    int choice, key, i, j, k, len;
    char text[100], result[100];
    char rail[20][100];
    FILE *input, *output;
    printf("--- Rail Fence Cipher ---\n1. Encrypt\n2. Decrypt\nChoice: ");
    scanf("%d", &choice);
    printf("Rails: "); scanf("%d", &key);
    input = fopen("input.txt", "r");
    if (!input) { printf("File not found!"); return 0; }
    output = fopen("output.txt", "w");
    while (fgets(text, sizeof(text), input))
    {
        text[strcspn(text, "\n")] = '\0';
        if (!strlen(text)) continue;
        len = strlen(text);
        if (choice == 1)
        {
            for(i=0;i<key;i++) for(j=0;j<len;j++) rail[i][j]='\n';
            i=0; k=1;
            for(j=0;j<len;j++)
            {
                rail[i][j]=text[j];
                if(i==key-1) k=-1;
                else if(i==0) k=1;
                i+=k;
            }
            k=0;
            for(i=0;i<key;i++) for(j=0;j<len;j++)
                if(rail[i][j]!='\n') result[k++]=rail[i][j];
            result[k]='\0';
        }
        else if (choice == 2)
        {
            for(i=0;i<key;i++) for(j=0;j<len;j++) rail[i][j]='\n';
            i=0; k=1;
            for(j=0;j<len;j++)
            {
                rail[i][j]='*';
                if(i==key-1) k=-1;
                else if(i==0) k=1;
                i+=k;
            }
            k=0;
            for(i=0;i<key;i++) for(j=0;j<len;j++)
                if(rail[i][j]=='*') rail[i][j]=text[k++];
            k=0; i=0; j=1;
            while(k<len)
            {
                result[k++]=rail[i][k-1];
                if(i==key-1) j=-1;
                else if(i==0) j=1;
                i+=j;
            }
            result[k]='\0';
        }
        else { printf("Invalid!"); return 0; }
        fprintf(output, "%s\n", result);
    }
    fclose(input); fclose(output);
    printf("Done! Check output.txt\n");
    return 0;
}
