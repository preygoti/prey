/* ==========================================================
 * CODE NO 4
 * HEADING : HILL CIPHER (2x2 or 3x3 key matrix)
 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
 
/* throw away whatever is left on the current input line */
static void clear_line(void)
{
    int c;
    while ((c = getchar()) != '
' && c != EOF)
        ;

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '';
    buf[strcspn(buf, "

")] = '';
}
 
int main()
{
    int choice;
    char line[2000], cleanText[2000], resultText[2000];
    int matrix[10][10], invMatrix[10][10];
    int n, i, j, k, cleanLen = 0;
 
    printf("
--- HILL CIPHER ---
");
    printf("1. Encrypt
");
    printf("2. Decrypt
");
    printf("Select an option (1 or 2): ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    {
        clear_line();
        printf("Invalid choice!
");
        return 0;
    }
    clear_line();
 
    /* --- STEP 1: READ TEXT FROM TERMINAL --- */
    read_line("Enter text : ", line, sizeof line);
 
    for (i = 0; line[i] != '' && cleanLen < 1990; i++)
    {
        if (isalpha((unsigned char)line[i]))
            cleanText[cleanLen++] = (char)toupper((unsigned char)line[i]);
    }
    cleanText[cleanLen] = '';
 
    if (cleanLen == 0)
    {
        printf("
[ERROR] No alphabetic letters found in the text!
");
        return 0;
    }
    printf("
Cleaned Input Text: %s
", cleanText);
 
    /* --- STEP 2: INPUT MATRIX SIZE & ELEMENTS --- */
    printf("Enter matrix size N (2 or 3): ");
    if (scanf("%d", &n) != 1 || (n != 2 && n != 3))
    {
        clear_line();
        printf("[ERROR] This version supports N = 2 or N = 3.
");
        return 0;
    }
 
    printf("
Enter %dx%d Key Matrix elements row by row:
", n, n);
    for (i = 0; i < n; i++)
    {
        printf("Row %d values (%d numbers): ", i + 1, n);
        for (j = 0; j < n; j++)
        {
            if (scanf("%d", &matrix[i][j]) != 1)
            {
                clear_line();
                printf("[ERROR] Matrix elements must be integers!
");
                return 0;
            }
            matrix[i][j] = (matrix[i][j] % 26 + 26) % 26; /* range 0-25 */
        }
    }
    clear_line();
 
    /* --- STEP 3: CHECK THE KEY (needed for BOTH encrypt and decrypt) --- */
    int det = 0;
    if (n == 2)
    {
        det = matrix[0][0] * matrix[1][1] - matrix[0][1] * matrix[1][0];
    }
    else
    {
        det = matrix[0][0] * (matrix[1][1] * matrix[2][2] - matrix[1][2] * matrix[2][1]) -
              matrix[0][1] * (matrix[1][0] * matrix[2][2] - matrix[1][2] * matrix[2][0]) +
              matrix[0][2] * (matrix[1][0] * matrix[2][1] - matrix[1][1] * matrix[2][0]);
    }
    det = (det % 26 + 26) % 26;
 
    int detInv = -1;
    for (int x = 1; x < 26; x++)
    {
        if ((det * x) % 26 == 1)
        {
            detInv = x;
            break;
        }
    }
    if (detInv == -1)
    {
        printf("
[ERROR] Key matrix is not invertible modulo 26! (gcd(det, 26) != 1)
");
        printf("Text encrypted with this key could never be decrypted. Choose another key.
");
        return 0;
    }
 
    /* Pad text with 'X' so length is a multiple of N */
    while (cleanLen % n != 0)
        cleanText[cleanLen++] = 'X';
    cleanText[cleanLen] = '';
    printf("Padded Text for Processing: %s
", cleanText);
 
    /* --- STEP 4: ENCRYPTION / DECRYPTION --- */
    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += matrix[j][k] * val;
                }
                resultText[resIdx++] = (char)(((sum % 26) + 26) % 26 + 'A');
            }
        }
        resultText[resIdx] = '';
 
        printf("
-------------------------
");
        printf("Encryption Successful!
");
        printf("Generated Ciphertext: %s
", resultText);
        printf("-------------------------
");
    }
    else
    {
        /* -------- DECRYPTION -------- */
        /* Adjugate matrix * det^-1  (mod 26) */
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
            {
                int subDet = 0;
                if (n == 2)
                {
                    int r = (i == 0) ? 1 : 0;
                    int c = (j == 0) ? 1 : 0;
                    subDet = matrix[r][c];
                }
                else
                {
                    int sub[2][2];
                    int r_idx = 0;
                    for (int row = 0; row < 3; row++)
                    {
                        if (row == i)
                            continue;
                        int c_idx = 0;
                        for (int col = 0; col < 3; col++)
                        {
                            if (col == j)
                                continue;
                            sub[r_idx][c_idx++] = matrix[row][col];
                        }
                        r_idx++;
                    }
                    subDet = sub[0][0] * sub[1][1] - sub[0][1] * sub[1][0];
                }
 
                int sign = ((i + j) % 2 == 0) ? 1 : -1;
                int cofactor = sign * subDet;
                cofactor = (cofactor % 26 + 26) % 26;
 
                /* transpose: store at [j][i] */
                invMatrix[j][i] = (cofactor * detInv) % 26;
            }
        }
 
        printf("
Calculated Inverse Key Matrix (mod 26):
");
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
                printf("%3d", invMatrix[i][j]);
            printf("
");
        }
 
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += invMatrix[j][k] * val;
                }
                resultText[resIdx++] = (char)(((sum % 26) + 26) % 26 + 'A');
            }
        }
        resultText[resIdx] = '';
 
        printf("
-------------------------
");
        printf("Decryption Successful!
");
        printf("Generated Plaintext: %s
", resultText);
        printf("-------------------------
");
    }
 
    return 0;
}

 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
 
#define MAX_LEN 1000
 
/* throw away whatever is left on the current input line */
static void clear_line(void)

    while ((c = getchar()) != '
' && c != EOF)
        ;

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '';
    buf[strcspn(buf, "

")] = '';
}
 
int main()
{
    char input_text[MAX_LEN]   = {0};
    char letters[MAX_LEN]      = {0};
    char initial_key[MAX_LEN]  = {0};
    char adjusted_key[MAX_LEN] = {0};
    char output_text[MAX_LEN]  = {0};
 
    int mode_choice, key_choice;
    int text_len, k_len, i, m, ki, nl = 0;
 
    printf("
--- VIGENERE / AUTOKEY CIPHER ---
");
    printf("1. Encrypt plaintext
");
    printf("2. Decrypt ciphertext
");
    printf("Enter operational choice (1 or 2): ");
    if (scanf("%d", &mode_choice) != 1 || (mode_choice != 1 && mode_choice != 2))
    {
        clear_line();
        printf("Invalid choice!
");
        return 0;
    }
    clear_line();
 
    read_line("Enter text : ", input_text, MAX_LEN);
    text_len = (int)strlen(input_text);
 
    if (text_len == 0)
    {
        printf("Error: text is empty.
");
        return 0;
    }
 
    /* collect only the letters of the text (spaces/digits do not use the key) */
    for (i = 0; i < text_len; i++)
    {
        if (isalpha((unsigned char)input_text[i]))
            letters[nl++] = input_text[i];
    }
 
    printf("Enter the secret key (letters only): ");
    if (scanf("%999s", initial_key) != 1)
    {
        clear_line();
        printf("Invalid key!
");
        return 0;
    }
    clear_line();
    k_len = (int)strlen(initial_key);
 
    for (i = 0; i < k_len; i++)
    {
        if (!isalpha((unsigned char)initial_key[i]))
        {
            printf("Error: key must contain letters only.
");
            return 0;
        }
        initial_key[i] = (char)toupper((unsigned char)initial_key[i]);
    }
 
    printf("
--- Keystream Adjustment Menu ---
");
    printf("1. Repeat the key to match length (Vigenere style)
");
    printf("2. Use Autokey style to match length
");
    printf("Enter choice (1 or 2): ");
    if (scanf("%d", &key_choice) != 1)
    {
        clear_line();
        printf("Invalid choice! Exiting program.
");
        return 0;
    }
    clear_line();
 
    /* keystream has one key letter for every LETTER of the text */
    if (key_choice == 1)
    {
        /* Option 1: key repeats */
        for (m = 0; m < nl; m++)
            adjusted_key[m] = initial_key[m % k_len];
    }
    else if (key_choice == 2)
    {
        /* Option 2: key first, then the plaintext itself */
        for (m = 0; m < nl && m < k_len; m++)
            adjusted_key[m] = initial_key[m];
 
        if (mode_choice == 1)
        {
            /* encryption: plaintext is already known */
            for (m = k_len; m < nl; m++)
                adjusted_key[m] = (char)toupper((unsigned char)letters[m - k_len]);
        }
        /* decryption: rest of the keystream is filled while decrypting */
    }
    else
    {
        printf("Invalid choice! Exiting program.
");
        return 0;
    }
    adjusted_key[nl] = '';
 
    /* Process the text (Encrypt or Decrypt) */
    ki = 0; /* keystream position - moves only on letters */
    for (i = 0; i < text_len; i++)
    {
        if (isalpha((unsigned char)input_text[i]))
        {
            char t_base = isupper((unsigned char)input_text[i]) ? 'A' : 'a';
            int t_val = input_text[i] - t_base;
            int k_val = adjusted_key[ki] - 'A';
 
            if (mode_choice == 1)
            {
                /* Encryption: (P + K) % 26 */
                output_text[i] = (char)(((t_val + k_val) % 26) + t_base);
            }
            else
            {
                /* Decryption: (C - K + 26) % 26 */
                output_text[i] = (char)(((t_val - k_val + 26) % 26) + t_base);
 
                /* Autokey: recovered plaintext letter becomes future key */
                if (key_choice == 2 && ki + k_len < nl)
                    adjusted_key[ki + k_len] = (char)toupper((unsigned char)output_text[i]);
            }
            ki++;
        }
        else
        {
            /* keep symbols, numbers and spaces as they are */
            output_text[i] = input_text[i];
        }
    }
    output_text[text_len] = '';
 
    if (mode_choice == 1)
        printf("
Ciphertext: %s

[ERROR] Key matrix is not invertible modulo 26! (gcd(det, 26) != 1)
");
        printf("Text encrypted with this key could never be decrypted. Choose another key.
");
        return 0;
    }
 
    /* Pad text with 'X' so length is a multiple of N */
    while (cleanLen % n != 0)
        cleanText[cleanLen++] = 'X';
    cleanText[cleanLen] = '';
    printf("Padded Text for Processing: %s
", cleanText);
 
    /* --- STEP 4: ENCRYPTION / DECRYPTION --- */
    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += matrix[j][k] * val;
                }
                resultText[resIdx++] = (char)(((sum % 26) + 26) % 26 + 'A');
            }
        }
        resultText[resIdx] = '';
 
        outFp = fopen(HILL_OUT, "w");
        if (outFp != NULL)
        {
            fputs(resultText, outFp);
            fclose(outFp);
        }
        else
        {
            printf("[ERROR] Could not create '%s'!
", HILL_OUT);
            return 0;
        }
 
        printf("
-------------------------
");
        printf("Encryption Successful!
");
        printf("Generated Ciphertext: %s
", resultText);
        printf("Result saved to '%s'
", HILL_OUT);
        printf("-------------------------
");
    }
    else
    {
        /* -------- DECRYPTION -------- */
        /* Adjugate matrix * det^-1  (mod 26) */
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
            {
                int subDet = 0;
                if (n == 2)
                {
                    int r = (i == 0) ? 1 : 0;
                    int c = (j == 0) ? 1 : 0;
                    subDet = matrix[r][c];
                }
                else
                {
                    int sub[2][2];
                    int r_idx = 0;
                    for (int row = 0; row < 3; row++)
                    {
                        if (row == i)
                            continue;
                        int c_idx = 0;
                        for (int col = 0; col < 3; col++)
                        {
                            if (col == j)
                                continue;
                            sub[r_idx][c_idx++] = matrix[row][col];
                        }
                        r_idx++;
                    }
                    subDet = sub[0][0] * sub[1][1] - sub[0][1] * sub[1][0];
                }
 
                int sign = ((i + j) % 2 == 0) ? 1 : -1;
                int cofactor = sign * subDet;
                cofactor = (cofactor % 26 + 26) % 26;
 
                /* transpose: store at [j][i] */
                invMatrix[j][i] = (cofactor * detInv) % 26;
            }
        }
 
        printf("
Calculated Inverse Key Matrix (mod 26):
");
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
                printf("%3d", invMatrix[i][j]);
            printf("
");
        }
 
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += invMatrix[j][k] * val;
                }
                resultText[resIdx++] = (char)(((sum % 26) + 26) % 26 + 'A');
            }
        }
        resultText[resIdx] = '';
 
        outFp = fopen(HILL_OUT, "w");
        if (outFp != NULL)
        {
            fputs(resultText, outFp);
            fclose(outFp);
        }
        else
        {
            printf("[ERROR] Could not create '%s'!
", HILL_OUT);
            return 0;
        }
 
        printf("
-------------------------
");
        printf("Decryption Successful!
");
        printf("Generated Plaintext: %s
", resultText);
        printf("Result saved to '%s'
", HILL_OUT);
        printf("-------------------------
");
    }
 
    return 0;
}

int main()
{
    char input_text[MAX_LEN]   = {0};
    char letters[MAX_LEN]      = {0};
    char initial_key[MAX_LEN]  = {0};
    char adjusted_key[MAX_LEN] = {0};
    char output_text[MAX_LEN]  = {0};
 
    int mode_choice, key_choice;
    int text_len, k_len, i, m, ki, nl = 0;

--- VIGENERE / AUTOKEY CIPHER ---
");
    printf("1. Encrypt plaintext
");
    printf("2. Decrypt ciphertext
");

        clear_line();

")] = 0; /* remove trailing newline */

    if (text_len == 0)
    {
        printf("Error: '%s' is empty.
", VIG_IN);
        return 0;
    }
 
    /* collect only the letters of the text (spaces/digits do not use the key) */
    for (i = 0; i < text_len; i++)
    {
        if (isalpha((unsigned char)input_text[i]))
            letters[nl++] = input_text[i];
    }
 
    printf("
Text read from file: %s
", input_text);
    printf("Enter the secret key (letters only): ");
    if (scanf("%999s", initial_key) != 1)
    {
        clear_line();
        printf("Invalid key!
");
        return 0;
    }
    clear_line();
    k_len = (int)strlen(initial_key);
 
    for (i = 0; i < k_len; i++)
    {
        if (!isalpha((unsigned char)initial_key[i]))
        {
            printf("Error: key must contain letters only.
");
            return 0;
        }
        initial_key[i] = (char)toupper((unsigned char)initial_key[i]);
    }
 
    printf("
--- Keystream Adjustment Menu ---
");
    printf("1. Repeat the key to match length (Vigenere style)
");
    printf("2. Use Autokey style to match length

    {
        clear_line();
        printf("Invalid choice! Exiting program.
");
        return 0;
    }
    clear_line();
 
    /* keystream has one key letter for every LETTER of the text */
    if (key_choice == 1)
    {
        /* Option 1: key repeats */
        for (m = 0; m < nl; m++)
            adjusted_key[m] = initial_key[m % k_len];
    }
    else if (key_choice == 2)
    {
        /* Option 2: key first, then the plaintext itself */
        for (m = 0; m < nl && m < k_len; m++)
            adjusted_key[m] = initial_key[m];
 
        if (mode_choice == 1)
        {
            /* encryption: plaintext is already known */
            for (m = k_len; m < nl; m++)
                adjusted_key[m] = (char)toupper((unsigned char)letters[m - k_len]);
        }
        /* decryption: rest of the keystream is filled while decrypting */
    }
    else
    {
        printf("Invalid choice! Exiting program.
");
        return 0;
    }
    adjusted_key[nl] = '';
 
    /* Process the text (Encrypt or Decrypt) */
    ki = 0; /* keystream position - moves only on letters */
    for (i = 0; i < text_len; i++)
    {
        if (isalpha((unsigned char)input_text[i]))
        {
            char t_base = isupper((unsigned char)input_text[i]) ? 'A' : 'a';
            int t_val = input_text[i] - t_base;
            int k_val = adjusted_key[ki] - 'A';
 
            if (mode_choice == 1)
            {
                /* Encryption: (P + K) % 26 */
                output_text[i] = (char)(((t_val + k_val) % 26) + t_base);
            }
            else
            {
                /* Decryption: (C - K + 26) % 26 */
                output_text[i] = (char)(((t_val - k_val + 26) % 26) + t_base);
 
                /* Autokey: recovered plaintext letter becomes future key */
                if (key_choice == 2 && ki + k_len < nl)
                    adjusted_key[ki + k_len] = (char)toupper((unsigned char)output_text[i]);
            }
            ki++;
        }
        else
        {
            /* keep symbols, numbers and spaces as they are */
            output_text[i] = input_text[i];
        }
    }
    output_text[text_len] = '';
 
    /* Save the outputs into the output file */
    fout = fopen(VIG_OUT, "w");
    if (fout == NULL)
    {
        printf("Error: Could not create '%s'.
", VIG_OUT);
        return 0;
    }
 
    if (mode_choice == 1)
        fprintf(fout, "Ciphertext: %s
", output_text);
    else
        fprintf(fout, "Plaintext: %s
", output_text);
 
    fprintf(fout, "Used Keystream: %s
", adjusted_key);
    fclose(fout);
 
    printf("
Operation successful! Results saved to '%s'.
", VIG_OUT);
 
    return 0;
}
