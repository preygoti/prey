/* ==========================================================
 * CODE NO 1
 * HEADING : CAESAR CIPHER
 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_LEN 1000

/* throw away whatever is left on the current input line */
static void clear_line(void)
{
    int c;
    while ((c = getchar()) != '\n' && c != EOF)
        ;
}

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '\0';
    buf[strcspn(buf, "\r\n")] = '\0';
}

int main()
{
    int choice, key, i, ch;
    char text[MAX_LEN], result[MAX_LEN];

    printf("\n--- CAESAR CIPHER ---");
    printf("\n1) Encrypt (plain text ---> cipher text)");
    printf("\n2) Decrypt (cipher text ---> plain text)");
    printf("\n3) Exit");
    printf("\nSelect your choice : ");
    if (scanf("%d", &choice) != 1)
    {
        clear_line();
        printf("Invalid input!\n");
        return 0;
    }

    if (choice == 3)
    {
        clear_line();
        printf("You are exiting....\n");
        return 0;
    }
    if (choice != 1 && choice != 2)
    {
        clear_line();
        printf("Please select proper choice\n");
        return 0;
    }

    printf("Enter your key (1-25) : ");
    if (scanf("%d", &key) != 1 || key < 1 || key > 25)
    {
        clear_line();
        printf("Key must be a number between 1 and 25!\n");
        return 0;
    }
    clear_line();

    read_line("Enter text : ", text, MAX_LEN);

    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */
        for (i = 0; text[i] != '\0'; i++)
        {
            ch = (unsigned char)text[i];
            if (isalpha(ch))
            {
                char base = isupper(ch) ? 'A' : 'a';
                int n = ch - base;
                n = (n + key) % 26;
                ch = n + base;
            }
            result[i] = (char)ch;
        }
        result[i] = '\0';
        printf("\nCipher Text : %s\n", result);
    }
    else
    {
        /* -------- DECRYPTION -------- */
        for (i = 0; text[i] != '\0'; i++)
        {
            ch = (unsigned char)text[i];
            if (isalpha(ch))
            {
                char base = isupper(ch) ? 'A' : 'a';
                int n = ch - base;
                n = (n - key + 26) % 26;
                ch = n + base;
            }
            result[i] = (char)ch;
        }
        result[i] = '\0';
        printf("\nPlain Text : %s\n", result);
    }

    return 0;
}

{

    const char KEY[]      = "MUALVOZKRNJXQDFSHPEBCTIWYG";
    int choice, i, j, ch, index, is_lower;
    char text[MAX_LEN], result[MAX_LEN], up, out;

    printf("\n--- MONOALPHABETIC CIPHER ---");

    {
        printf("\nPlease select proper option\n");
        return 0;
    }

    read_line("Enter text : ", text, MAX_LEN);

    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */
        for (i = 0; text[i] != '\0'; i++)
        {
            ch = (unsigned char)text[i];
            if (isalpha(ch))
            {
                is_lower = islower(ch);
                up = (char)toupper(ch);

                index = 0;
                for (j = 0; j < 26; j++)
                {
                    if (ALPHABET[j] == up)
                    {
                        index = j;
                        break;
                    }
                }
                out = KEY[index];
                ch = is_lower ? tolower(out) : out;
            }
            result[i] = (char)ch;
        }
        result[i] = '\0';
        printf("\nCipher Text : %s\n", result);
    }
    else
    {
        /* -------- DECRYPTION -------- */
        for (i = 0; text[i] != '\0'; i++)
        {
            ch = (unsigned char)text[i];
            if (isalpha(ch))
            {
                is_lower = islower(ch);
                up = (char)toupper(ch);

                index = 0;
                for (j = 0; j < 26; j++)
                {
                    if (KEY[j] == up)
                    {
                        index = j;
                        break;
                    }
                }
                out = ALPHABET[index];
                ch = is_lower ? tolower(out) : out;
            }
            result[i] = (char)ch;
        }
        result[i] = '\0';
        printf("\nPlain Text : %s\n", result);
    }

    return 0;
}


/* ==========================================================
 * CODE NO 3
 * HEADING : PLAYFAIR CIPHER
 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

/* throw away whatever is left on the current input line */
static void clear_line(void)
{
    int c;
    while ((c = getchar()) != '\n' && c != EOF)
        ;
}

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '\0';
    buf[strcspn(buf, "\r\n")] = '\0';
}

int main()
{
    int choice;
    char key[100], text[500];
    char matrix[5][5];
    int used[26] = {0};
    int i, j, k = 0;

    printf("\n--- PLAYFAIR CIPHER ---\n");
    printf("1. Encrypt plaintext\n");
    printf("2. Decrypt ciphertext\n");
    printf("Enter operational choice (1 or 2): ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    {
        clear_line();
        printf("Invalid choice!\n");
        return 0;
    }

    printf("Enter Key (one word, letters only) : ");
    if (scanf("%99s", key) != 1)
    {
        clear_line();
        printf("Invalid key!\n");
        return 0;
    }
    clear_line();

    used['J' - 'A'] = 1; /* I and J share one cell */

    /* Generate Matrix */
    for (i = 0; key[i] != '\0'; i++)
    {
        char ch = (char)toupper((unsigned char)key[i]);

        if (ch == 'J')
            ch = 'I';

        if (ch >= 'A' && ch <= 'Z')
        {
            if (used[ch - 'A'] == 0)
            {
                matrix[k / 5][k % 5] = ch;
                used[ch - 'A'] = 1;
                k++;
            }
        }
    }

    for (i = 0; i < 26; i++)
    {
        if (used[i] == 0)
        {
            matrix[k / 5][k % 5] = (char)(i + 'A');
            k++;
        }
    }

    printf("\nKey Matrix\n\n");
    for (i = 0; i < 5; i++)
    {
        for (j = 0; j < 5; j++)
            printf("%c ", matrix[i][j]);
        printf("\n");
    }

    if (choice == 1)
        read_line("\nEnter Plain Text : ", text, 500);
    else
        read_line("\nEnter Cipher Text : ", text, 500);

    /* keep only letters, uppercase, J -> I */
    char clean[500];
    int p = 0;

    for (i = 0; text[i] != '\0'; i++)
    {
        if (isalpha((unsigned char)text[i]))
        {
            clean[p] = (char)toupper((unsigned char)text[i]);
            if (clean[p] == 'J')
                clean[p] = 'I';
            p++;
        }
    }
    clean[p] = '\0';

    if (p == 0)
    {
        printf("No letters found in the text!\n");
        return 0;
    }

    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */

        /* Make pairs: if both letters of a pair are equal, put a filler
           between them; if one letter is left at the end, add a filler. */
        char final[1100];
        int f = 0;

        i = 0;
        while (i < p)
        {
            final[f++] = clean[i];

            if (i + 1 < p && clean[i] != clean[i + 1])
            {
                final[f++] = clean[i + 1];
                i += 2;
            }
            else
            {
                final[f++] = (clean[i] == 'X') ? 'Q' : 'X';
                i += 1;
            }
        }
        final[f] = '\0';

        printf("\nPlain Text (prepared) : %s\n", final);

        char cipher[1100];
        int c = 0;

        for (i = 0; i < f; i += 2)
        {
            char a = final[i];
            char b = final[i + 1];
            int r1 = 0, c1 = 0, r2 = 0, c2 = 0;

            for (int x = 0; x < 5; x++)
            {
                for (int y = 0; y < 5; y++)
                {
                    if (matrix[x][y] == a) { r1 = x; c1 = y; }
                    if (matrix[x][y] == b) { r2 = x; c2 = y; }
                }
            }

            if (r1 == r2)
            {
                cipher[c++] = matrix[r1][(c1 + 1) % 5];
                cipher[c++] = matrix[r2][(c2 + 1) % 5];
            }
            else if (c1 == c2)
            {
                cipher[c++] = matrix[(r1 + 1) % 5][c1];
                cipher[c++] = matrix[(r2 + 1) % 5][c2];
            }
            else
            {
                cipher[c++] = matrix[r1][c2];
                cipher[c++] = matrix[r2][c1];
            }
        }
        cipher[c] = '\0';

        printf("Cipher Text : %s\n", cipher);
    }
    else
    {
        /* -------- DECRYPTION -------- */

        /* a valid Playfair cipher text is made of pairs of DIFFERENT letters */
        if (p % 2 != 0)
        {
            printf("Error: cipher text must have an even number of letters.\n");
            return 0;
        }
        for (i = 0; i < p; i += 2)
        {
            if (clean[i] == clean[i + 1])
            {
                printf("Error: invalid cipher text (a pair has two identical letters).\n");
                return 0;
            }
        }

        char decrypt[1100];
        int d = 0;

        for (i = 0; i < p; i += 2)
        {
            char a = clean[i];
            char b = clean[i + 1];
            int r1 = 0, c1 = 0, r2 = 0, c2 = 0;

            for (int x = 0; x < 5; x++)
            {
                for (int y = 0; y < 5; y++)
                {
                    if (matrix[x][y] == a) { r1 = x; c1 = y; }
                    if (matrix[x][y] == b) { r2 = x; c2 = y; }
                }
            }

            if (r1 == r2)
            {
                decrypt[d++] = matrix[r1][(c1 + 4) % 5];
                decrypt[d++] = matrix[r2][(c2 + 4) % 5];
            }
            else if (c1 == c2)
            {
                decrypt[d++] = matrix[(r1 + 4) % 5][c1];
                decrypt[d++] = matrix[(r2 + 4) % 5][c2];
            }
            else
            {
                decrypt[d++] = matrix[r1][c2];
                decrypt[d++] = matrix[r2][c1];
            }
        }
        decrypt[d] = '\0';

        printf("\nPlain Text : %s\n", decrypt);
        printf("(X or Q filler letters, if any, are padding)\n");
    }

    return 0;
}


/* ==========================================================
 * CODE NO 4
 * HEADING : HILL CIPHER (2x2 or 3x3 key matrix)
 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

/* throw away whatever is left on the current input line */
static void clear_line(void)
{
    int c;
    while ((c = getchar()) != '\n' && c != EOF)
        ;
}

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '\0';
    buf[strcspn(buf, "\r\n")] = '\0';
}

int main()
{
    int choice;
    char line[2000], cleanText[2000], resultText[2000];
    int matrix[10][10], invMatrix[10][10];
    int n, i, j, k, cleanLen = 0;

    printf("\n--- HILL CIPHER ---\n");
    printf("1. Encrypt\n");
    printf("2. Decrypt\n");
    printf("Select an option (1 or 2): ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    {
        clear_line();
        printf("Invalid choice!\n");
        return 0;
    }
    clear_line();

    /* --- STEP 1: READ TEXT FROM TERMINAL --- */
    read_line("Enter text : ", line, sizeof line);

    for (i = 0; line[i] != '\0' && cleanLen < 1990; i++)
    {
        if (isalpha((unsigned char)line[i]))
            cleanText[cleanLen++] = (char)toupper((unsigned char)line[i]);
    }
    cleanText[cleanLen] = '\0';

    if (cleanLen == 0)
    {
        printf("\n[ERROR] No alphabetic letters found in the text!\n");
        return 0;
    }
    printf("\nCleaned Input Text: %s\n", cleanText);

    /* --- STEP 2: INPUT MATRIX SIZE & ELEMENTS --- */
    printf("Enter matrix size N (2 or 3): ");
    if (scanf("%d", &n) != 1 || (n != 2 && n != 3))
    {
        clear_line();
        printf("[ERROR] This version supports N = 2 or N = 3.\n");
        return 0;
    }

    printf("\nEnter %dx%d Key Matrix elements row by row:\n", n, n);
    for (i = 0; i < n; i++)
    {
        printf("Row %d values (%d numbers): ", i + 1, n);
        for (j = 0; j < n; j++)
        {
            if (scanf("%d", &matrix[i][j]) != 1)
            {
                clear_line();
                printf("[ERROR] Matrix elements must be integers!\n");
                return 0;
            }
            matrix[i][j] = (matrix[i][j] % 26 + 26) % 26; /* range 0-25 */
        }
    }
    clear_line();

    /* --- STEP 3: CHECK THE KEY (needed for BOTH encrypt and decrypt) --- */
    int det = 0;
    if (n == 2)
    {
        det = matrix[0][0] * matrix[1][1] - matrix[0][1] * matrix[1][0];
    }
    else
    {
        det = matrix[0][0] * (matrix[1][1] * matrix[2][2] - matrix[1][2] * matrix[2][1]) -
              matrix[0][1] * (matrix[1][0] * matrix[2][2] - matrix[1][2] * matrix[2][0]) +
              matrix[0][2] * (matrix[1][0] * matrix[2][1] - matrix[1][1] * matrix[2][0]);
    }
    det = (det % 26 + 26) % 26;

    int detInv = -1;
    for (int x = 1; x < 26; x++)
    {
        if ((det * x) % 26 == 1)
        {
            detInv = x;
            break;
        }
    }
    if (detInv == -1)
    {
        printf("\n[ERROR] Key matrix is not invertible modulo 26! (gcd(det, 26) != 1)\n");
        printf("Text encrypted with this key could never be decrypted. Choose another key.\n");
        return 0;
    }

    /* Pad text with 'X' so length is a multiple of N */
    while (cleanLen % n != 0)
        cleanText[cleanLen++] = 'X';
    cleanText[cleanLen] = '\0';
    printf("Padded Text for Processing: %s\n", cleanText);

    /* --- STEP 4: ENCRYPTION / DECRYPTION --- */
    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += matrix[j][k] * val;
                }
                resultText[resIdx++] = (char)(((sum % 26) + 26) % 26 + 'A');
            }
        }
        resultText[resIdx] = '\0';

        printf("\n-------------------------\n");
        printf("Encryption Successful!\n");
        printf("Generated Ciphertext: %s\n", resultText);
        printf("-------------------------\n");
    }
    else
    {
        /* -------- DECRYPTION -------- */
        /* Adjugate matrix * det^-1  (mod 26) */
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
            {
                int subDet = 0;
                if (n == 2)
                {
                    int r = (i == 0) ? 1 : 0;
                    int c = (j == 0) ? 1 : 0;
                    subDet = matrix[r][c];
                }
                else
                {
                    int sub[2][2];
                    int r_idx = 0;
                    for (int row = 0; row < 3; row++)
                    {
                        if (row == i)
                            continue;
                        int c_idx = 0;
                        for (int col = 0; col < 3; col++)
                        {
                            if (col == j)
                                continue;
                            sub[r_idx][c_idx++] = matrix[row][col];
                        }
                        r_idx++;
                    }
                    subDet = sub[0][0] * sub[1][1] - sub[0][1] * sub[1][0];
                }

                int sign = ((i + j) % 2 == 0) ? 1 : -1;
                int cofactor = sign * subDet;
                cofactor = (cofactor % 26 + 26) % 26;

                /* transpose: store at [j][i] */
                invMatrix[j][i] = (cofactor * detInv) % 26;
            }
        }

        printf("\nCalculated Inverse Key Matrix (mod 26):\n");
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
                printf("%3d", invMatrix[i][j]);
            printf("\n");
        }

        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    int val = cleanText[i + k] - 'A';
                    sum += invMatrix[j][k] * val;
                }
                resultText[resIdx++] = (char)(((sum % 26) + 26) % 26 + 'A');
            }
        }
        resultText[resIdx] = '\0';

        printf("\n-------------------------\n");
        printf("Decryption Successful!\n");
        printf("Generated Plaintext: %s\n", resultText);
        printf("-------------------------\n");
    }

    return 0;
}


/* ==========================================================
 * CODE NO 5
 * HEADING : VIGENERE CIPHER (with Autokey option)
 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_LEN 1000

/* throw away whatever is left on the current input line */
static void clear_line(void)
{
    int c;
    while ((c = getchar()) != '\n' && c != EOF)
        ;
}

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '\0';
    buf[strcspn(buf, "\r\n")] = '\0';
}

int main()
{
    char input_text[MAX_LEN]   = {0};
    char letters[MAX_LEN]      = {0};
    char initial_key[MAX_LEN]  = {0};
    char adjusted_key[MAX_LEN] = {0};
    char output_text[MAX_LEN]  = {0};

    int mode_choice, key_choice;
    int text_len, k_len, i, m, ki, nl = 0;

    printf("\n--- VIGENERE / AUTOKEY CIPHER ---\n");
    printf("1. Encrypt plaintext\n");
    printf("2. Decrypt ciphertext\n");
    printf("Enter operational choice (1 or 2): ");
    if (scanf("%d", &mode_choice) != 1 || (mode_choice != 1 && mode_choice != 2))
    {
        clear_line();
        printf("Invalid choice!\n");
        return 0;
    }
    clear_line();

    read_line("Enter text : ", input_text, MAX_LEN);
    text_len = (int)strlen(input_text);

    if (text_len == 0)
    {
        printf("Error: text is empty.\n");
        return 0;
    }

    /* collect only the letters of the text (spaces/digits do not use the key) */
    for (i = 0; i < text_len; i++)
    {
        if (isalpha((unsigned char)input_text[i]))
            letters[nl++] = input_text[i];
    }

    printf("Enter the secret key (letters only): ");
    if (scanf("%999s", initial_key) != 1)
    {
        clear_line();
        printf("Invalid key!\n");
        return 0;
    }
    clear_line();
    k_len = (int)strlen(initial_key);

    for (i = 0; i < k_len; i++)
    {
        if (!isalpha((unsigned char)initial_key[i]))
        {
            printf("Error: key must contain letters only.\n");
            return 0;
        }
        initial_key[i] = (char)toupper((unsigned char)initial_key[i]);
    }

    printf("\n--- Keystream Adjustment Menu ---\n");
    printf("1. Repeat the key to match length (Vigenere style)\n");
    printf("2. Use Autokey style to match length\n");
    printf("Enter choice (1 or 2): ");
    if (scanf("%d", &key_choice) != 1)
    {
        clear_line();
        printf("Invalid choice! Exiting program.\n");
        return 0;
    }
    clear_line();

    /* keystream has one key letter for every LETTER of the text */
    if (key_choice == 1)
    {
        /* Option 1: key repeats */
        for (m = 0; m < nl; m++)
            adjusted_key[m] = initial_key[m % k_len];
    }
    else if (key_choice == 2)
    {
        /* Option 2: key first, then the plaintext itself */
        for (m = 0; m < nl && m < k_len; m++)
            adjusted_key[m] = initial_key[m];

        if (mode_choice == 1)
        {
            /* encryption: plaintext is already known */
            for (m = k_len; m < nl; m++)
                adjusted_key[m] = (char)toupper((unsigned char)letters[m - k_len]);
        }
        /* decryption: rest of the keystream is filled while decrypting */
    }
    else
    {
        printf("Invalid choice! Exiting program.\n");
        return 0;
    }
    adjusted_key[nl] = '\0';

    /* Process the text (Encrypt or Decrypt) */
    ki = 0; /* keystream position - moves only on letters */
    for (i = 0; i < text_len; i++)
    {
        if (isalpha((unsigned char)input_text[i]))
        {
            char t_base = isupper((unsigned char)input_text[i]) ? 'A' : 'a';
            int t_val = input_text[i] - t_base;
            int k_val = adjusted_key[ki] - 'A';

            if (mode_choice == 1)
            {
                /* Encryption: (P + K) % 26 */
                output_text[i] = (char)(((t_val + k_val) % 26) + t_base);
            }
            else
            {
                /* Decryption: (C - K + 26) % 26 */
                output_text[i] = (char)(((t_val - k_val + 26) % 26) + t_base);

                /* Autokey: recovered plaintext letter becomes future key */
                if (key_choice == 2 && ki + k_len < nl)
                    adjusted_key[ki + k_len] = (char)toupper((unsigned char)output_text[i]);
            }
            ki++;
        }
        else
        {
            /* keep symbols, numbers and spaces as they are */
            output_text[i] = input_text[i];
        }
    }
    output_text[text_len] = '\0';

    if (mode_choice == 1)
        printf("\nCiphertext: %s\n", output_text);
    else
        printf("\nPlaintext: %s\n", output_text);

    printf("Used Keystream: %s\n", adjusted_key);

    return 0;
}


/* ==========================================================
 * CODE NO 6
 * HEADING : COLUMNAR TRANSPOSITION CIPHER (keyed)
 * Columns are read in the alphabetical order of the key letters.
 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_LEN 1000

/* throw away whatever is left on the current input line */
static void clear_line(void)
{
    int c;
    while ((c = getchar()) != '\n' && c != EOF)
        ;
}

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '\0';
    buf[strcspn(buf, "\r\n")] = '\0';
}

int main()
{
    int choice, i, j, k, rows, n, len, a, b, tmp;
    char text[MAX_LEN + 20], key[20], result[MAX_LEN + 20];
    char matrix[MAX_LEN + 20][20];
    int order[20];

    printf("\n--- COLUMNAR TRANSPOSITION CIPHER ---\n");
    printf("1. Encrypt plaintext\n");
    printf("2. Decrypt ciphertext\n");
    printf("Enter operational choice (1 or 2): ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    {
        clear_line();
        printf("Invalid choice!\n");
        return 0;
    }

    printf("Enter the secret key (max 19 characters): ");
    if (scanf("%19s", key) != 1)
    {
        clear_line();
        printf("Invalid key!\n");
        return 0;
    }
    clear_line();

    n = (int)strlen(key);

    /* column order: sort column numbers by key letter (ties keep position) */
    for (i = 0; i < n; i++)
        order[i] = i;
    for (a = 0; a < n - 1; a++)
    {
        for (b = 0; b < n - 1 - a; b++)
        {
            if (toupper((unsigned char)key[order[b]]) > toupper((unsigned char)key[order[b + 1]]))
            {
                tmp = order[b];
                order[b] = order[b + 1];
                order[b + 1] = tmp;
            }
        }
    }

    read_line("Enter text : ", text, MAX_LEN);
    len = (int)strlen(text);

    if (len == 0)
    {
        printf("Error: text is empty.\n");
        return 0;
    }

    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */
        while (len % n != 0)
            text[len++] = 'X';
        text[len] = '\0';

        rows = len / n;

        k = 0;
        for (i = 0; i < rows; i++)
            for (j = 0; j < n; j++)
                matrix[i][j] = text[k++];

        k = 0;
        for (j = 0; j < n; j++)
            for (i = 0; i < rows; i++)
                result[k++] = matrix[i][order[j]];
        result[k] = '\0';

        printf("\nCipher Text : %s\n", result);
    }
    else
    {
        /* -------- DECRYPTION -------- */
        if (len % n != 0)
        {
            printf("Error: ciphertext length must be a multiple of the key length (%d).\n", n);
            return 0;
        }

        rows = len / n;

        k = 0;
        for (j = 0; j < n; j++)
            for (i = 0; i < rows; i++)
                matrix[i][order[j]] = text[k++];

        k = 0;
        for (i = 0; i < rows; i++)
            for (j = 0; j < n; j++)
                result[k++] = matrix[i][j];
        result[k] = '\0';

        printf("\nPlain Text : %s\n", result);
        printf("(trailing X letters, if any, are padding)\n");
    }

    return 0;
}


/* ==========================================================
 * CODE NO 7
 * HEADING : RAIL FENCE TRANSPOSITION CIPHER
 * ========================================================== */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_LEN 1000

/* throw away whatever is left on the current input line */
static void clear_line(void)
{
    int c;
    while ((c = getchar()) != '\n' && c != EOF)
        ;
}

/* print a prompt, read one full line (spaces allowed), remove the newline */
static void read_line(const char *prompt, char *buf, int size)
{
    printf("%s", prompt);
    if (fgets(buf, size, stdin) == NULL)
        buf[0] = '\0';
    buf[strcspn(buf, "\r\n")] = '\0';
}

int main()
{
    int choice, key, i, j, k, d, len;
    char text[MAX_LEN], result[MAX_LEN];
    char rail[20][MAX_LEN];

    printf("\n--- RAIL FENCE TRANSPOSITION CIPHER ---\n");
    printf("1. Encrypt plaintext\n");
    printf("2. Decrypt ciphertext\n");
    printf("Enter operational choice (1 or 2): ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    {
        clear_line();
        printf("Invalid choice!\n");
        return 0;
    }

    printf("Enter number of rails (2-20): ");
    if (scanf("%d", &key) != 1 || key < 2 || key > 20)
    {
        clear_line();
        printf("Number of rails must be between 2 and 20!\n");
        return 0;
    }
    clear_line();

    read_line("Enter text : ", text, MAX_LEN);
    len = (int)strlen(text);

    if (len == 0)
    {
        printf("Error: text is empty.\n");
        return 0;
    }

    for (i = 0; i < key; i++)
        for (j = 0; j < len; j++)
            rail[i][j] = '\0';

    if (choice == 1)
    {
        /* -------- ENCRYPTION -------- */
        i = 0;
        d = 1;
        for (j = 0; j < len; j++)
        {
            rail[i][j] = text[j];

            if (i == key - 1)
                d = -1;
            else if (i == 0)
                d = 1;

            i = i + d;
        }

        k = 0;
        for (i = 0; i < key; i++)
            for (j = 0; j < len; j++)
                if (rail[i][j] != '\0')
                    result[k++] = rail[i][j];
        result[k] = '\0';

        printf("\nCipher Text : %s\n", result);
    }
    else
    {
        /* -------- DECRYPTION -------- */
        /* 1. mark the zig-zag path with '*' */
        i = 0;
        d = 1;
        for (j = 0; j < len; j++)
        {
            rail[i][j] = '*';

            if (i == key - 1)
                d = -1;
            else if (i == 0)
                d = 1;

            i = i + d;
        }

        /* 2. fill the marked cells row by row with the cipher text */
        k = 0;
        for (i = 0; i < key; i++)
            for (j = 0; j < len; j++)
                if (rail[i][j] == '*')
                    rail[i][j] = text[k++];

        /* 3. read the zig-zag path again */
        i = 0;
        d = 1;
        for (j = 0; j < len; j++)
        {
            result[j] = rail[i][j];

            if (i == key - 1)
                d = -1;
            else if (i == 0)
                d = 1;

            i = i + d;
        }
        result[len] = '\0';

        printf("\nPlain Text : %s\n", result);
    }

    return 0;
}
