    if (choice == 3) return 0;
    if (choice != 1 && choice != 2) {
        printf("please select proper choice\n");

    printf("enter your text: ");

    text[strcspn(text, "\r\n")] = '\0';

    if (choice == 1)
    {
        for (i = 0; text[i] != '\0'; i++)
        {
            char ch = text[i];
            if (isalpha(ch))
            {
                char base = isupper(ch) ? 'A' : 'a';
                int n = ch - base;
                n = (n + key) % 26;
                text[i] = n + base;
            }
        }
        printf("\nCipher Text: %s\n", text);
        printf("done!!!!!!!!!!!!!!!!!!!!\n");
    }
    else if (choice == 2)
    {
        for (i = 0; text[i] != '\0'; i++)
        {
            char ch = text[i];
            if (isalpha(ch))
            {
                char base = isupper(ch) ? 'A' : 'a';
                int n = ch - base;
                n = (n - key + 26) % 26;
                text[i] = n + base;
            }
        }
        printf("\nPlain Text: %s\n", text);
        printf("done!!!!!!!!!!!!!!!!!!!!\n");
    }
    return 0;
}

/* ============================================================
   PRACTICAL 3 : Playfair Cipher
   ============================================================ */
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main()
{
    char key[100], text[500];
    char matrix[5][5];
    int used[26] = {0};
    int i, j, k = 0;

    printf("Enter Key: ");
    scanf("%s", key);
    used['J' - 'A'] = 1;

    // Generate Matrix
    for (i = 0; key[i] != '\0'; i++)
    {
        char ch = toupper(key[i]);
        if (ch == 'J') ch = 'T';
        if (ch >= 'A' && ch <= 'Z' && used[ch - 'A'] == 0)
        {
            matrix[k / 5][k % 5] = ch;
            used[ch - 'A'] = 1;
            k++;
        }
    }
    for (i = 0; i < 26; i++)
    {
        if (used[i] == 0)
        {
            matrix[k / 5][k % 5] = i + 'A';
            k++;
        }
    }

    printf("\nKey Matrix:\n");
    for (i = 0; i < 5; i++)
    {
        for (j = 0; j < 5; j++) printf("%c ", matrix[i][j]);
        printf("\n");
    }

    printf("\nEnter Plain Text: ");
    getchar();
    fgets(text, sizeof(text), stdin);
    text[strcspn(text, "\r\n")] = '\0';

    char plain[500];
    int p = 0;
    for (i = 0; text[i] != '\0'; i++)
    {
        if (isalpha(text[i]))
        {
            plain[p] = toupper(text[i]);
            if (plain[p] == 'J') plain[p] = 'T';
            p++;
        }
    }
    plain[p] = '\0';

    char final[500];
    int f = 0;
    for (i = 0; i < p; i++)
    {
        final[f++] = plain[i];
        if (i < p - 1 && plain[i] == plain[i + 1])
            final[f++] = 'X';
    }
    if (f % 2 != 0) final[f++] = 'X';
    final[f] = '\0';

    printf("\nPrepared Text: %s\n", final);

    // Encryption
    char cipher[500];
    int c = 0;
    for (i = 0; i < f; i += 2)
    {
        char a = final[i], b = final[i + 1];
        int r1, c1, r2, c2;
        for (int x = 0; x < 5; x++)
        {
            for (int y = 0; y < 5; y++)
            {
                if (matrix[x][y] == a) { r1 = x; c1 = y; }
                if (matrix[x][y] == b) { r2 = x; c2 = y; }
            }
        }
        if (r1 == r2)
        {
            cipher[c++] = matrix[r1][(c1 + 1) % 5];
            cipher[c++] = matrix[r2][(c2 + 1) % 5];
        }
        else if (c1 == c2)
        {
            cipher[c++] = matrix[(r1 + 1) % 5][c1];
            cipher[c++] = matrix[(r2 + 1) % 5][c2];
        }
        else
        {
            cipher[c++] = matrix[r1][c2];
            cipher[c++] = matrix[r2][c1];
        }
    }
    cipher[c] = '\0';
    printf("Cipher Text: %s\n", cipher);

    // Decryption
    char decrypt[500];
    int d = 0;
    for (i = 0; i < c; i += 2)
    {
        char a = cipher[i], b = cipher[i + 1];
        int r1, c1, r2, c2;
        for (int x = 0; x < 5; x++)
        {
            for (int y = 0; y < 5; y++)
            {
                if (matrix[x][y] == a) { r1 = x; c1 = y; }
                if (matrix[x][y] == b) { r2 = x; c2 = y; }
            }
        }
        if (r1 == r2)
        {
            decrypt[d++] = matrix[r1][(c1 + 4) % 5];
            decrypt[d++] = matrix[r2][(c2 + 4) % 5];
        }
        else if (c1 == c2)
        {
            decrypt[d++] = matrix[(r1 + 4) % 5][c1];
            decrypt[d++] = matrix[(r2 + 4) % 5][c2];
        }
        else
        {
            decrypt[d++] = matrix[r1][c2];
            decrypt[d++] = matrix[r2][c1];
        }
    }
    decrypt[d] = '\0';
    printf("Decrypted Text: %s\n", decrypt);
    return 0;
}


/* ============================================================
   PRACTICAL 4 : Hill Cipher
   ============================================================ */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int main()
{
    int choice;
    char text[2000], cleanText[2000], resultText[2000];
    int matrix[10][10], invMatrix[10][10];
    int n, i, j, k, cleanLen = 0;

    printf("=== HILL CIPHER PROGRAM ===\n1. Encrypt\n2. Decrypt\nSelect: ");
    if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2))
    {
        printf("Invalid choice!\n");
        return 1;
    }

    printf("Enter Text: ");
    getchar();
    fgets(text, sizeof(text), stdin);

    for (i = 0; text[i] != '\0'; i++)
    {
        if (isalpha(text[i]))
            cleanText[cleanLen++] = toupper((unsigned char)text[i]);
    }
    cleanText[cleanLen] = '\0';

    if (cleanLen == 0) {
        printf("[ERROR] No valid alphabetic letters!\n");
        return 1;
    }

    printf("Enter matrix size N (2 or 3): ");
    if (scanf("%d", &n) != 1 || (n != 2 && n != 3))
    {
        printf("Supports N=2 or N=3 only.\n");
        return 1;
    }

    printf("Enter %dx%d Key Matrix elements row by row:\n", n, n);
    for (i = 0; i < n; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &matrix[i][j]);
            matrix[i][j] = (matrix[i][j] % 26 + 26) % 26;
        }
    }

    while (cleanLen % n != 0)
    {
        cleanText[cleanLen++] = 'X';
    }
    cleanText[cleanLen] = '\0';

    if (choice == 1)
    {
        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    sum += matrix[j][k] * (cleanText[i + k] - 'A');
                }
                resultText[resIdx++] = ((sum % 26) + 26) % 26 + 'A';
            }
        }
        resultText[resIdx] = '\0';
        printf("\nCipher Text: %s\n", resultText);
    }
    else if (choice == 2)
    {
        int det = 0;
        if (n == 2)
        {
            det = matrix[0][0] * matrix[1][1] - matrix[0][1] * matrix[1][0];
        }
        else if (n == 3)
        {
            det = matrix[0][0] * (matrix[1][1] * matrix[2][2] - matrix[1][2] * matrix[2][1]) -
                  matrix[0][1] * (matrix[1][0] * matrix[2][2] - matrix[1][2] * matrix[2][0]) +
                  matrix[0][2] * (matrix[1][0] * matrix[2][1] - matrix[1][1] * matrix[2][0]);
        }
        det = (det % 26 + 26) % 26;

        int detInv = -1;
        for (int x = 1; x < 26; x++)
        {
            if ((det * x) % 26 == 1) { detInv = x; break; }
        }
        if (detInv == -1)
        {
            printf("\n[ERROR] Key matrix is not invertible modulo 26!\n");
            return 1;
        }

        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
            {
                int subDet = 0;
                if (n == 2)
                {
                    int r = (i == 0) ? 1 : 0;
                    int c = (j == 0) ? 1 : 0;
                    subDet = matrix[r][c];
                }
                else if (n == 3)
                {
                    int sub[2][2], r_idx = 0;
                    for (int row = 0; row < 3; row++)
                    {
                        if (row == i) continue;
                        int c_idx = 0;
                        for (int col = 0; col < 3; col++)
                        {
                            if (col == j) continue;
                            sub[r_idx][c_idx++] = matrix[row][col];
                        }
                        r_idx++;
                    }
                    subDet = sub[0][0] * sub[1][1] - sub[0][1] * sub[1][0];
                }
                int sign = ((i + j) % 2 == 0) ? 1 : -1;
                int cofactor = (sign * subDet % 26 + 26) % 26;
                invMatrix[j][i] = (cofactor * detInv) % 26;
            }
        }

        int resIdx = 0;
        for (i = 0; i < cleanLen; i += n)
        {
            for (j = 0; j < n; j++)
            {
                int sum = 0;
                for (k = 0; k < n; k++)
                {
                    sum += invMatrix[j][k] * (cleanText[i + k] - 'A');
                }
                resultText[resIdx++] = ((sum % 26) + 26) % 26 + 'A';
            }
        }
        resultText[resIdx] = '\0';
        printf("\nDecrypted Text: %s\n", resultText);
    }
    return 0;
}


/* ============================================================
   PRACTICAL 5 : Vigenere Cipher (with Autokey option)
   ============================================================ */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#define MAX_LEN 1000

int main()
{
    char input_text[MAX_LEN] = {0};
    char initial_key[MAX_LEN] = {0};
    char adjusted_key[MAX_LEN] = {0};
    char output_text[MAX_LEN] = {0};
    int mode_choice, key_choice;
    int text_len, k_len, i;

    printf("--- Cryptography System ---\n1. Encrypt\n2. Decrypt\nChoice: ");
    if (scanf("%d", &mode_choice) != 1) return 1;

    printf("Enter Text: ");
    getchar();
    fgets(input_text, MAX_LEN, stdin);
    input_text[strcspn(input_text, "\r\n")] = '\0';
    text_len = strlen(input_text);

    if (text_len == 0) {
        printf("Error: Text is empty.\n");
        return 1;
    }

    printf("Enter the secret key: ");
    scanf("%s", initial_key);
    k_len = strlen(initial_key);

    printf("1. Repeat the key to match length (Vigenere style)\n2. Use Autokey style to match length\nEnter choice (1 or 2): ");
    scanf("%d", &key_choice);

    if (key_choice == 1)
    {
        for (i = 0; i < text_len; i++)
            adjusted_key[i] = initial_key[i % k_len];
        adjusted_key[text_len] = '\0';
    }
    else if (key_choice == 2)
    {
        for (i = 0; i < k_len; i++)
            adjusted_key[i] = initial_key[i];
        if (mode_choice == 1)
        {
            for (i = k_len; i < text_len; i++)
                adjusted_key[i] = input_text[i - k_len];
        }
        else
        {
            for (i = k_len; i < text_len; i++)
                adjusted_key[i] = '\0';
        }
        adjusted_key[text_len] = '\0';
    }
    else
    {
        printf("Invalid choice!\n");
        return 1;
    }

    for (i = 0; i < text_len; i++)
    {
        if (isalpha(input_text[i]))
        {
            char t_base = isupper(input_text[i]) ? 'A' : 'a';
            char k_base = isupper(adjusted_key[i]) ? 'A' : 'a';
            int t_val = input_text[i] - t_base;
            int k_val = adjusted_key[i] - k_base;

            if (mode_choice == 1)
            {
                output_text[i] = ((t_val + k_val) % 26) + t_base;
            }
            else
            {
                output_text[i] = ((t_val - k_val + 26) % 26) + t_base;
                if (key_choice == 2 && (i + k_len < text_len))
                    adjusted_key[i + k_len] = output_text[i];
            }
        }
        else
        {
            output_text[i] = input_text[i];
            if (mode_choice == 2 && key_choice == 2 && (i + k_len < text_len))
                adjusted_key[i + k_len] = input_text[i];
        }
    }
    output_text[text_len] = '\0';

    if (mode_choice == 1)
        printf("\nCiphertext: %s\n", output_text);
    else
        printf("\nPlaintext: %s\n", output_text);

    printf("Used Keystream: %s\n", adjusted_key);
    return 0;
}


/* ============================================================

   ============================================================ */
#include <stdio.h>
#include <string.h>

int main()
{
    int choice, i, j, k, rows, n;
    char text[100], key[20], result[100];
    char matrix[100][20];

    printf("--- Columnar Transposition ---\n1. Encrypt\n2. Decrypt\nChoice: ");
    scanf("%d", &choice);

    printf("Enter Key: ");
    scanf("%s", key);
    n = strlen(key);

    printf("Enter Text: ");
    getchar();
    fgets(text, sizeof(text), stdin);
    text[strcspn(text, "\r\n")] = '\0';

    if (choice == 1)
    {
        while (strlen(text) % n != 0) strcat(text, "X");
        rows = strlen(text) / n;
        k = 0;
        for (i = 0; i < rows; i++)
            for (j = 0; j < n; j++) matrix[i][j] = text[k++];
        k = 0;
        for (j = 0; j < n; j++)
            for (i = 0; i < rows; i++) result[k++] = matrix[i][j];
        result[k] = '\0';
        printf("\nCipher Text: %s\n", result);
    }
    else if (choice == 2)
    {
        rows = strlen(text) / n;
        k = 0;
        for (j = 0; j < n; j++)
            for (i = 0; i < rows; i++) matrix[i][j] = text[k++];
        k = 0;
        for (i = 0; i < rows; i++)
            for (j = 0; j < n; j++) result[k++] = matrix[i][j];
        result[k] = '\0';
        printf("\nPlain Text: %s\n", result);
    }
    else
    {
        printf("Invalid choice!\n");
    }
    return 0;
}


/* ============================================================
   PRACTICAL 7 : Rail Fence Transposition Cipher
   ============================================================ */
#include <stdio.h>
#include <string.h>

int main()
{
    int choice, key, i, j, k, len;
    char text[100], result[100];
    char rail[20][100];

    printf("--- Rail Fence Cipher ---\n1. Encrypt\n2. Decrypt\nChoice: ");
    scanf("%d", &choice);

    printf("Enter number of rails: ");
    scanf("%d", &key);

    printf("Enter Text: ");
    getchar();
    fgets(text, sizeof(text), stdin);
    text[strcspn(text, "\r\n")] = '\0';
    len = strlen(text);

    if (choice == 1)
    {
        for (i = 0; i < key; i++)
            for (j = 0; j < len; j++) rail[i][j] = '\n';

        i = 0; k = 1;
        for (j = 0; j < len; j++)
        {
            rail[i][j] = text[j];
            if (i == key - 1) k = -1;
            else if (i == 0) k = 1;
            i += k;
        }

        k = 0;
        for (i = 0; i < key; i++)
            for (j = 0; j < len; j++)
                if (rail[i][j] != '\n') result[k++] = rail[i][j];
        result[k] = '\0';
        printf("\nCipher Text: %s\n", result);
    }
    else if (choice == 2)
    {
        for (i = 0; i < key; i++)
            for (j = 0; j < len; j++) rail[i][j] = '\n';

        i = 0; k = 1;
        for (j = 0; j < len; j++)
        {
            rail[i][j] = '*';
            if (i == key - 1) k = -1;
            else if (i == 0) k = 1;
            i += k;
        }

        k = 0;
        for (i = 0; i < key; i++)
            for (j = 0; j < len; j++)
                if (rail[i][j] == '*') rail[i][j] = text[k++];

        k = 0; i = 0; j = 1;
        while (k < len)
        {
            result[k++] = rail[i][k - 1];
            if (i == key - 1) j = -1;
            else if (i == 0) j = 1;
            i += j;
        }
        result[k] = '\0';
        printf("\nPlain Text: %s\n", result);
    }
    else
    {
        printf("Invalid choice!\n");
    }
    return 0;
}
